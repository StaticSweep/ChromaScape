# API Contract

```yaml
openapi: 3.1.0
info:
  title: Replay Observed Playback API
  version: 1.0.0
paths:
  /api/replay/playback/v1/recordings:
    get:
      summary: List replayable observed recordings in stable keyset order
      security: [{ replayPrincipal: [] }]
      parameters:
        - { name: accountName, in: query, schema: { type: string } }
        - { name: limit, in: query, schema: { type: integer, default: 50, minimum: 1, maximum: 200 } }
        - { name: cursor, in: query, schema: { type: string } }
      responses:
        "200":
          description: Catalog page
          content:
            application/json:
              schema: { $ref: "#/components/schemas/CatalogPage" }
        "400": { $ref: "#/components/responses/ValidationError" }
  /api/replay/playback/v1/recordings/{recordingId}/frames:
    get:
      summary: Read captured frames and logs without invoking simulation
      security: [{ replayPrincipal: [] }]
      parameters:
        - { name: recordingId, in: path, required: true, schema: { type: string, minLength: 1 } }
        - { name: fromTick, in: query, schema: { type: integer, default: 0, minimum: 0 } }
        - { name: limit, in: query, schema: { type: integer, default: 100, minimum: 1, maximum: 500 } }
      responses:
        "200":
          description: Frame page
          content:
            application/json:
              schema: { $ref: "#/components/schemas/FramePage" }
        "400": { $ref: "#/components/responses/ValidationError" }
        "404":
          description: RECORDING_NOT_FOUND or DEBUG_SESSION_NOT_FOUND
components:
  securitySchemes:
    replayPrincipal: { type: apiKey, in: cookie, name: session }
  responses:
    ValidationError:
      description: VALIDATION_ERROR
  schemas:
    CatalogPage:
      type: object
      required: [schema, entries, hasMore]
      properties:
        schema: { const: replay-playback-catalog/v1 }
        entries:
          type: array
          items: { $ref: "#/components/schemas/CatalogEntry" }
        nextCursor: { type: [string, "null"] }
        hasMore: { type: boolean }
    CatalogEntry:
      type: object
      required: [recordingId, sessionId, startTime, tickCount, positionFrameCount, logCount, positionCoveragePct, replayable, replayabilityReason]
      properties:
        recordingId: { type: string }
        sessionId: { type: string }
        accountName: { type: [string, "null"] }
        scriptName: { type: [string, "null"] }
        startTime: { type: string, format: date-time }
        endTime: { type: [string, "null"], format: date-time }
        startTick: { type: [integer, "null"] }
        endTick: { type: [integer, "null"] }
        tickCount: { type: integer }
        positionFrameCount: { type: integer }
        logCount: { type: integer }
        positionCoveragePct: { type: number }
        replayable: { type: boolean }
        replayabilityReason: { type: string }
    FramePage:
      type: object
      required: [schema, recordingId, sessionId, frames, hasMore]
      properties:
        schema: { const: replay-playback-frame-page/v1 }
        recordingId: { type: string }
        sessionId: { type: string }
        accountName: { type: [string, "null"] }
        startTick: { type: [integer, "null"] }
        endTick: { type: [integer, "null"] }
        frames:
          type: array
          items: { $ref: "#/components/schemas/Frame" }
        nextFromTick: { type: [integer, "null"] }
        hasMore: { type: boolean }
    Frame:
      type: object
      required: [tick, playerState, inventory, equipment, threats, projectiles, cooldowns, quest, questHelper, npcs, gameObjects, groundObjects, groundItems, players, widgets, schemaVersion, logs]
      properties:
        tick: { type: integer }
        timestamp: { type: [string, "null"], format: date-time }
        playerState: { type: object, additionalProperties: true }
        inventory: { type: array }
        equipment: { type: array }
        threats: { type: array }
        projectiles: { type: array }
        cooldowns: { type: object }
        quest: { type: object }
        questHelper: { type: object, additionalProperties: true }
        npcs: { type: array }
        gameObjects: { type: array }
        groundObjects: { type: array }
        groundItems: { type: array }
        players: { type: array }
        widgets: { type: array }
        schemaVersion: { type: integer }
        logs:
          type: array
          items: { $ref: "#/components/schemas/TickLog" }
    TickLog:
      type: object
      required: [id, tick, subsystem, severity, eventType, message, details, categories]
      properties:
        id: { type: integer }
        tick: { type: integer }
        timestamp: { type: [string, "null"], format: date-time }
        subsystem: { type: string }
        severity: { type: string }
        eventType: { type: string }
        message: { type: string }
        details: { type: object, additionalProperties: true }
        correlationId: { type: [string, "null"] }
        categories:
          type: array
          items: { enum: [ACTION, DIALOGUE, VARBIT] }
```

## Compatibility and behavior

Unversioned playback aliases return the exact v1 schema. Only the deprecated alias accepts offset. The legacy tick-card endpoint returns `Deprecation: true` and a `Link` to v1. Cursor payload is URL-safe unpadded Base64 JSON `{version,startTime,id}`; invalid payloads are `VALIDATION_ERROR`. Frame logs are deterministic and always arrays. Empty frame pages are successful.
