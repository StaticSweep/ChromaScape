# Current-build API contract

```yaml
openapi: 3.1.0
info:
  title: Luna Max current-build evidence boundary
  version: 1.0.0
  description: Read-only capture, replay, shadow, and certified-candidate evidence.
servers:
  - url: http://localhost:8090
paths:
  /api/debug/builder-goals/launch-account:
    post:
      summary: Launch an explicitly configured local client
      description: Uses only explicit clean worktree roots and validates contract identity before launch.
      responses:
        "200": {description: Launch accepted}
        "409": {description: Contract, account, or gate mismatch; fail closed}
  /api/debug/builder-goals/machines:
    get:
      summary: List local launch hosts and identity readiness
      responses:
        "200": {description: Machine inventory}
  /api/debug/ai-routine/runs/from-stabilization/{attemptId}:
    post:
      summary: Create a managed capture only from verified stabilization provenance
      parameters:
        - in: path
          name: attemptId
          required: true
          schema: {type: string}
      responses:
        "201": {description: Managed run with MICRO6_MANAGED identity}
        "409": {description: Missing or mismatched provenance}
  /api/debug/ai-routine/captures:
    post:
      summary: Record one human-gated or managed capture
      responses:
        "201": {description: Recording identity and safety fields}
        "409": {description: Sequence, identity, or contract mismatch}
  /api/debug/ai-routine/replay:
    post:
      summary: Deterministically replay a recorded graph/model
      responses:
        "200": {description: Read-only replay evidence}
        "409": {description: Identity mismatch}
  /api/debug/ai-routine/shadow:
    post:
      summary: Run read-only shadow inference
      responses:
        "200": {description: Shadow evidence with control_allowed false}
        "409": {description: Unsafe or stale input}
components:
  schemas:
    Safety:
      type: object
      required: [control_allowed, control_dispatch_count]
      properties:
        control_allowed: {type: boolean, const: false}
        control_dispatch_count: {type: integer, const: 0}
    ClientIdentity:
      type: object
      required: [client_id, client_kind, client_build_id, artifact_digest, adapter_contract]
      properties:
        client_id: {type: string}
        client_kind: {type: string, enum: [MICRO6_MANAGED, RUNELITE_CAPTURE]}
        client_build_id: {type: string}
        artifact_digest: {type: string}
        adapter_contract: {type: string}
    Candidate:
      type: object
      required: [candidate_id, candidate_hash, certification_hash, signature]
      properties:
        candidate_id: {type: string}
        candidate_hash: {type: string}
        certification_hash: {type: string}
        signature: {type: string}
